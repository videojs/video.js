### Selector

<FrameworkCase frameworks={["react"]}>
Pass `selectControls` to <DocsLink slug="reference/api/use-player">`usePlayer`</DocsLink> to subscribe to controls state. Returns `undefined` if the controls feature is not configured.
</FrameworkCase>

<FrameworkCase frameworks={["html"]}>
Pass `selectControls` to <DocsLink slug="reference/api/player-controller">`PlayerController`</DocsLink> to subscribe to controls state. Returns `undefined` if the controls feature is not configured.
</FrameworkCase>

<FrameworkCase frameworks={["react"]}>
```tsx title="ControlsOverlay.tsx"
import { selectControls, usePlayer } from '@videojs/react';

function ControlsOverlay({ children }: { children: React.ReactNode }) {
  const controls = usePlayer(selectControls);
  if (!controls) return null;

  return (
    <div style={{ opacity: controls.controlsVisible ? 1 : 0 }}>
      {children}
    </div>
  );
}
```
</FrameworkCase>

<FrameworkCase frameworks={["html"]}>
```ts title="controls-overlay.ts"
import { createPlayer, UIElement, selectControls } from '@videojs/html';
import { videoFeatures } from '@videojs/html/video';

const { PlayerController } = createPlayer({ features: videoFeatures });

class ControlsOverlay extends UIElement {
  readonly #controls = new PlayerController(this, selectControls);
}
```
</FrameworkCase>
